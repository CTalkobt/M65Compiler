; test_tune_trace_xemu: tunes whose register writes the real player and
; tune45's model must agree on, frame by frame

instrument lead  pulse 09 a6 0600 2 1 vdelay=6
instrument sweep pulse 00 f0 0700 0 0 pwm=40
instrument kick  pulse 00 f0 0800 0 0 slide=-3
instrument chip  pulse 00 f0 0400 0 0
instrument hat   noise 00 a0 0800 0 1

; every effect: vibrato after a delay, a pulse sweep, a pitch slide, an
; arpeggio, ties, a note longer than 255 ticks, 2.5 frames a tick
tune effects 2.5 once
v1: @lead c5:4 e5:4 ~:4 g5:2 eb5:2 | c6:300 r:6
v2: @sweep c3:16 g2:8 ~:8 r:4 c3:286
v3: @chip arp:4,7 c4:8 arp:3,7 a3:8 arp:off @kick c2:2 r:2 c2:2 r:2 @hat c7:1 r:1 c7:1 r:1 \
    @chip arp:5,9 c4:294

; glides: up and down, slow and fast, held on by a tie, with vibrato (the
; trace starts 40 ticks in: tune_seek)
tune glides 1 once
v1: @chip c4^3:12 ~:8 g4^-5:20 c5^12,1:40 r:4 e4^2,7:6 r:110
v2: @lead c3^7:30 c3:10 eb3^-3,2:60 r:100
v3: @kick c2^-30,2:8 r:8 c2^-30,2:8 r:8 @hat c6^1:8 @chip arp:4,7 c4^5:20 arp:off c4:140

; filter, wave tables and hard restart: two filtered instruments on one chip
; (a cutoff sweep up to its end, one down), an unfiltered one taking its
; voice out; a wave table held at its last step and one that loops, with
; a glide and an arpeggio; hard restart of 1 and 2 frames before notes,
; not before ties or rests, after instrument changes, across the loop;
; patterns: called once and twice, an instrument change inside one, the
; hard restart looking into one, round its repeat, out of it and across the
; loop into one; a legato note (the pitch, the note going on); a tail
; instrument (its two-note arpeggio on through the release), pw: and cut:;
; pwrun (the pulse width on across notes and rests) and cutrun (a note
; leaving the cutoff as cut: set it); patterns called transposed (a legato
; note in one too); a pulse-width cycle (its limits: 19-byte instruments), a sawtooth one
wavetable drum noise pulse:-12 pulse:-19 pulse:-24
wavetable buzz pulse saw pulse:12 saw:12 loop 2
instrument bass  pulse 00 f8 0400 0 0 hr=2 filter=lp res=12 cutoff=700 fsweep=40
instrument snare pulse 00 f0 0800 0 0 hr=1 table=drum
instrument slead saw 08 a8 0800 3 0 hr=2 table=buzz filter=bp+hp res=4 cutoff=500 fsweep=-9 vdelay=3 vspeed=3
instrument plain tri 00 f0 0800 0 0
instrument tl pulse 00 f9 0080 0 0 tail pwm=-60 free pwrun
instrument cr saw 00 f0 0800 0 0 filter=lp res=8 cutrun pwrun
instrument cy pulse 00 f0 0600 0 0 pwm=48 pwlo=5c0 pwhi=680
instrument cs pulse 00 f0 0600 0 0 pwm=48 pwlo=5c0 pwhi=640 pwsaw
tune synth 2.5 loop
pattern pb: c2:4
pattern ps: c5:1
pattern pv: @slead g4:6 ~:2
pattern pl: eb2:1 ~f2:1
v1: @bass *pb*2 |: tempo:2 ~:2 *pl+2 r:2 g1:2 @cr cut:300,5 c3:2 c3:2 @bass *pb
v2: @snare c5:2 r:2 *ps *ps-5 r:2 |: arp:3,7 c5:4 arp:off @cy e5:2 r:1 @cs e5:1 @snare c5:2 r:2 c5:4
v3: *pv |: @tl arp2:12 c5:1 pw:600,20 r:1 cut:300,-9 r:2 arp:off @slead g4^5,2:4 r:4 tempo:3.25 c4:4
